# Pinned. These were unpinned name-only entries, which meant a redeploy months
# from now could install a new major of any of them and fail in a way that had
# nothing to do with the commit that triggered it.
#
# Every pin must install on the Python that render.yaml asks for, currently
# 3.11. That is not automatic: these were first taken from a machine running
# 3.13, and one of them - numpy - had moved past 3.11 support, which failed the
# build rather than anything in the app.
flask==3.1.2
gunicorn==23.0.0
openai==2.29.0
python-dotenv==1.2.2
httpx==0.28.1
# Held below the local version on purpose. numpy 2.5 requires Python >= 3.12
# and Render builds this on 3.11, so pinning what happened to be installed
# locally broke the deploy with "No matching distribution found". 2.4.6
# supports >= 3.11, so it installs on both, and it was checked against the real
# index before pinning: same cosines, self-match 1.0.
#
# The lesson is that a pin taken from a developer machine is a guess about the
# build environment until something confirms it. Everything else here was
# already compatible; numpy was the only one that was not.
numpy==2.4.6
requests==2.34.2
beautifulsoup4==4.15.0

# Windows has no system timezone database, so zoneinfo cannot resolve
# America/Chicago there. feeds.py needs it to place calendar times, which
# Google publishes in UTC.
tzdata==2026.1

# Reads the PDFs the site links. Pure Python, no system libraries, which is
# what makes it safe on a free instance - see pdfs.py for what it is and is
# not asked to do.
pypdf==6.18.0
